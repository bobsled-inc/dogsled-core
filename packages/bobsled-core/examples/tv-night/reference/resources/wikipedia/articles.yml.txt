spec_version: "1"
type: resource
name: articles
integration: wikipedia
description: Short introductions for federation. Fetch full_article separately
  for evidence assessment.
input:
  titles:
    type: string
    default: The Good Place
api:
  endpoint: /w/api.php
  method: GET
  query_params:
    action: query
    format: json
    formatversion: "2"
    prop: extracts|revisions
    explaintext: "1"
    exchars: "1200"
    exlimit: "5"
    rvprop: ids
    titles: ${{ input.titles }}
    exintro: "1"
output:
  page_id:
    type: integer
    path: $.pageid
  title:
    type: string
    path: $.title
  revision_id:
    type: integer
    path: $.revisions[0].revid
  article_text:
    type: string
    path: $.extract
  requested_titles:
    type: string
    from:
      input: titles
extract:
  record_path: $.query.pages
  table:
    primary_key:
      - page_id
query:
  filters:
    requested_titles:
      operators:
        - eq
      bind:
        eq:
          input: titles
